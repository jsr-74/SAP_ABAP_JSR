MODULE SHOW_PASS_8100 OUTPUT.
  LOOP AT SCREEN.
    IF SCREEN-NAME = 'C_PASS' OR SCREEN-NAME = 'RE_PASS'.
      SCREEN-INVISIBLE  = '1'.
      ENDIF.

IF SHOW = 'X'.
  IF SCREEN-NAME = 'C_PASS' OR SCREEN-NAME = 'RE_PASS'.
      SCREEN-INVISIBLE  = '0'.

      ENDIF.
      ENDIF.
  MODIFY SCREEN.
    ENDLOOP.


ENDMODULE.

*&---------------------------------------------------------------------*
*&      Module  MODIFY_8011_SCREN  OUTPUT
*&---------------------------------------------------------------------*
*       text
*----------------------------------------------------------------------*
MODULE MODIFY_8011_SCREN OUTPUT.
IF RAD1 IS INITIAL AND RAD2 IS INITIAL AND RAD3 IS INITIAL.
  RAD1 = 'X'.
  ENDIF.

if rad1 = 'X'.

  LOOP AT SCREEN.

    IF SCREEN-GROUP1 = 'USB' ."OR  rad2 = 'X'.
      SCREEN-ACTIVE = '0'.
      ENDIF.
      MODIFY SCREEN.

  ENDLOOP.
  ELSEif rad2 = 'X'.

  LOOP AT SCREEN.

    IF SCREEN-GROUP1 = 'PSG'.
      SCREEN-ACTIVE = '0'.
      ENDIF.
      MODIFY SCREEN.

  ENDLOOP.
  ELSEif rad3 = 'X'.

  LOOP AT SCREEN.

    IF SCREEN-GROUP2 = 'U&P'.
      SCREEN-ACTIVE = '1'.
      ENDIF.
      MODIFY SCREEN.

  ENDLOOP.
  ENDIF.
ENDMODULE.
